def collatz_steps(n):
    steps = 0

        steps = steps + 1
    return steps

### `loops-while-3` — Угадать за семь попыток

Компьютер угадывает число от 1 до 100 делением диапазона пополам — тот самый алгоритм, который вы уже разбирали.

Напиши функцию, которая возвращает, сколько попыток на это уйдёт.

Правило строгое, иначе ответы разойдутся: каждая попытка — это `mid = (lo + hi) // 2`. Если не угадали, отбрасываем ненужную половину.

Скрытый тест: `binary_guess(75)` → `2`

**Python**

```python
def binary_guess(secret):
    lo = 1
    hi = 100
    tries = 0
    while True:
        mid = (lo + hi) // 2
        tries = tries + 1
        if mid == secret:
            return tries
        if mid < secret:
            lo = mid + 1
        else:
            hi = mid - 1
```

**C#**

```csharp
using System;
using System.Collections.Generic;

class Program
{
    static int BinaryGuess(int secret)
    {
        int lo = 1;
        int hi = 100;
        int tries = 0;
        while (true)
        {
            int mid = (lo + hi) / 2;
            tries = tries + 1;
            if (mid == secret)
            {
                return tries;
            }
            if (mid < secret)
            {
                lo = mid + 1;
            }
            else
            {
                hi = mid - 1;
            }
        }
    }
}
```

### `strings-3` — Шифр Цезаря

Продолжение вашего модуля «Этичный хакинг».

Напиши функцию, которая сдвигает каждую английскую букву на shift позиций по алфавиту. После «z» счёт идёт снова с «a».

Регистр сохраняется. Всё, что не английская буква — пробелы, запятые, цифры — остаётся как было.

Скрытый тест: `caesar("abc", 0)` → `"abc"`

**Python**

```python
def caesar(text, shift):
    result = ""
    for letter in text:
        if "a" <= letter <= "z":
            result = result + chr((ord(letter) - ord("a") + shift) % 26 + ord("a"))
        elif "A" <= letter <= "Z":
            result = result + chr((ord(letter) - ord("A") + shift) % 26 + ord("A"))
        else:
            result = result + letter
    return result
```

**C#**

```csharp
using System;
using System.Collections.Generic;

class Program
{
    static string Caesar(string text, int shift)
    {
        string result = "";
        foreach (char letter in text)
        {
            if (letter >= 'a' && letter <= 'z')
            {
                result = result + (char)((letter - 'a' + shift) % 26 + 'a');
            }
            else if (letter >= 'A' && letter <= 'Z')
            {
                result = result + (char)((letter - 'A' + shift) % 26 + 'A');
            }
            else
            {
                result = result + letter;
            }
        }
        return result;
    }
}
```

### `functions-3` — Надёжность пароля

Пароль проверяется по трём признакам, и каждый вынесен в свою функцию:

`is_long` — длина не меньше 8
`has_digit` — есть хотя бы одна цифра
`has_upper` — есть хотя бы одна большая английская буква

`password_strength` считает, сколько признаков выполнено: три — «сильный», два — «средний», меньше двух — «слабый». Она обязана вызывать все три функции.

Обязана вызывать: `is_long`, `has_digit`, `has_upper`.

Скрытый тест: `password_strength("ABC1")` → `"средний"`

**Python**

```python
def is_long(password):
    return len(password) >= 8


def has_digit(password):
    for ch in password:
        if ch in "0123456789":
            return True
    return False


def has_upper(password):
    for ch in password:
        if "A" <= ch <= "Z":
            return True
    return False


def password_strength(password):
    score = 0
    if is_long(password):
        score = score + 1
    if has_digit(password):
        score = score + 1
    if has_upper(password):
        score = score + 1
    if score == 3:
        return "сильный"
    if score == 2:
        return "средний"
    return "слабый"
```

**C#**

```csharp
using System;
using System.Collections.Generic;

class Program
{
    static bool IsLong(string password)
    {
        return password.Length >= 8;
    }

    static bool HasDigit(string password)
    {
        foreach (char ch in password)
        {
            if (ch >= '0' && ch <= '9')
            {
                return true;
            }
        }
        return false;
    }

    static bool HasUpper(string password)
    {
        foreach (char ch in password)
        {
            if (ch >= 'A' && ch <= 'Z')
            {
                return true;
            }
        }
        return false;
    }

    static string PasswordStrength(string password)
    {
        int score = 0;
        if (IsLong(password))
        {
            score = score + 1;
        }
        if (HasDigit(password))
        {
            score = score + 1;
        }
        if (HasUpper(password))
        {
            score = score + 1;
        }
        if (score == 3)
        {
            return "сильный";
        }
        if (score == 2)
        {
            return "средний";
        }
        return "слабый";
    }
}
```

### `dicts-1` — Сколько каких букв

Напиши функцию, которая считает, сколько раз встретился каждый символ, и возвращает словарь «символ → сколько раз».

Пробелы не считаются. Большие и маленькие буквы считаются одинаково, в ответе — маленькие.

Скрытый тест: `count_letters("")` → `{}`

**Python**

```python
def count_letters(text):
    count = {}
    for letter in text.lower():
        if letter != " ":
            count[letter] = count.get(letter, 0) + 1
    return count
```

**C#**

```csharp
using System;
using System.Collections.Generic;

class Program
{
    static Dictionary<string, int> CountLetters(string text)
    {
        Dictionary<string, int> count = new Dictionary<string, int>();
        foreach (char letter in text.ToLower())
        {
            if (letter != ' ')
            {
                string key = letter.ToString();
                if (count.ContainsKey(key))
                {
                    count[key] = count[key] + 1;
                }
                else
                {
                    count[key] = 1;
                }
            }
        }
        return count;
    }
}
```

### `dicts-2` — Лучший игрок

На вход приходит словарь «имя → счёт». Напиши функцию, которая возвращает имя игрока с наибольшим счётом.

Словарь всегда непустой, и лучший всегда один.

Скрытый тест: `best_player({"Аня": -2, "Боря": -5})` → `"Аня"`

**Python**

```python
def best_player(scores):
    best_name = None
    best_score = None
    for name, value in scores.items():
        if best_score is None or value > best_score:
            best_name = name
            best_score = value
    return best_name
```

**C#**

```csharp
using System;
using System.Collections.Generic;

class Program
{
    static string BestPlayer(Dictionary<string, int> scores)
    {
        string bestName = null;
        int bestScore = 0;
        bool first = true;
        foreach (var pair in scores)
        {
            if (first || pair.Value > bestScore)
            {
                bestName = pair.Key;
                bestScore = pair.Value;
                first = false;
            }
        }
        return bestName;
    }
}
```

### `dicts-3` — Сложить два счёта

Напиши функцию, которая объединяет два словаря «имя → счёт» в один.

Если имя встречается в обоих, счета складываются. Исходные словари менять нельзя.

Скрытый тест: `merge_scores({}, {"Ян": 9})` → `{"Ян": 9}`

**Python**

```python
def merge_scores(a, b):
    result = {}
    for name, value in a.items():
        result[name] = result.get(name, 0) + value
    for name, value in b.items():
        result[name] = result.get(name, 0) + value
    return result
```

**C#**

```csharp
using System;
using System.Collections.Generic;

class Program
{
    static Dictionary<string, int> MergeScores(Dictionary<string, int> a, Dictionary<string, int> b)
    {
        Dictionary<string, int> result = new Dictionary<string, int>();
        foreach (var pair in a)
        {
            result[pair.Key] = pair.Value;
        }
        foreach (var pair in b)
        {
            if (result.ContainsKey(pair.Key))
            {
                result[pair.Key] = result[pair.Key] + pair.Value;
            }
            else
            {
                result[pair.Key] = pair.Value;
            }
        }
        return result;
    }
}
```
